%% ------------------------------------------------------------------------
%% TikZ helper library for the Kyoto presentation theme: a thin styling
%% layer over ordinary TikZ (boxes, labels and arrows in the Kyoto
%% palette). It imposes no drawing DSL -- use ordinary TikZ geometry
%% (right=of, bend left/right, out=/in=, explicit coordinates, ...) and
%% apply the `kyoto/...' styles for appearance only. Works standalone
%% (falls back to its own copies of the Kyoto colors if the Beamer
%% theme has not defined them) and reuses the theme's palette without
%% redefinition when it has.
%% ------------------------------------------------------------------------
\ProvidesFile{tikzlibrarykyoto.code.tex}[2026-09-04 v1.2 Kyoto TikZ library]

\RequirePackage{xcolor}

\usetikzlibrary{arrows.meta}
\usetikzlibrary{positioning}
\usetikzlibrary{shapes.callouts}

%% ------------------------------------------------------------------
%% Palette: reuse the Beamer theme's colors if already defined;
%% otherwise define the same values as a standalone fallback.
%% \@ifundefinedcolor (from xcolor) is the correct "is this color name
%% already defined" test -- NOT \@ifundefined{color@Name}, which tests
%% xcolor's per-MODEL registry and would always report "undefined".
%% ------------------------------------------------------------------
\makeatletter
\@ifundefinedcolor{KyotoBg}{\definecolor{KyotoBg}{HTML}{EEEEEE}}{}
\@ifundefinedcolor{KyotoFg}{\definecolor{KyotoFg}{HTML}{222222}}{}
\@ifundefinedcolor{KyotoTheme}{\definecolor{KyotoTheme}{HTML}{081F57}}{}
\@ifundefinedcolor{KyotoEmph}{\definecolor{KyotoEmph}{HTML}{FF0027}}{}
\@ifundefinedcolor{KyotoLink}{\definecolor{KyotoLink}{HTML}{2D539E}}{}
\@ifundefinedcolor{KyotoSeparator}{\definecolor{KyotoSeparator}{HTML}{555555}}{}
\makeatother

%% ------------------------------------------------------------------
%% Public styles, namespaced under kyoto/. No fixed width/height, so
%% these compose with user-supplied `minimum width=', `text width=',
%% `align=', `right=of', etc.
%% ------------------------------------------------------------------
\tikzset{
  %% Boxes -- shared geometry, filled appearance by default -----------
  kyoto/box base/.style={
    rounded corners=3pt,
    line width=1pt,
    align=center,
    inner sep=8pt,
  },
  kyoto/box/.style={
    kyoto/box base,
    draw=KyotoTheme,
    fill=KyotoTheme,
    text=KyotoBg,
  },
  kyoto/box alert/.style={
    kyoto/box,
    draw=KyotoEmph,
    fill=KyotoEmph,
  },
  kyoto/box secondary/.style={
    kyoto/box,
    draw=KyotoSeparator,
    fill=KyotoSeparator,
  },
  %% Outline variants: light fill, colored border, dark text ----------
  kyoto/box outline/.style={
    kyoto/box base,
    draw=KyotoTheme,
    fill=KyotoBg,
    text=KyotoFg,
  },
  kyoto/box outline alert/.style={
    kyoto/box outline,
    draw=KyotoEmph,
  },
  kyoto/box outline secondary/.style={
    kyoto/box outline,
    draw=KyotoSeparator,
  },
  %% Callouts -- same filled-by-default / outline split as boxes ------
  kyoto/callout/.style={
    kyoto/box,
    rectangle callout,
  },
  kyoto/callout alert/.style={
    kyoto/box alert,
    rectangle callout,
  },
  kyoto/callout outline/.style={
    kyoto/box outline,
    rectangle callout,
  },
  kyoto/callout outline alert/.style={
    kyoto/box outline alert,
    rectangle callout,
  },
  %% Labels (text on/near arrows and objects) -------------------------
  kyoto/label/.style={
    text=KyotoFg,
  },
  %% Arrows ------------------------------------------------------------
  kyoto/arrow/.style={
    draw=KyotoTheme,
    line width=0.8pt,
    -{Stealth[length=2.5mm, width=2mm]},
  },
  kyoto/arrow alert/.style={
    kyoto/arrow,
    draw=KyotoEmph,
  },
  kyoto/arrow secondary/.style={
    kyoto/arrow,
    draw=KyotoSeparator,
  },
  kyoto/arrow dashed/.style={
    kyoto/arrow,
    dashed,
  },
}

\endinput
